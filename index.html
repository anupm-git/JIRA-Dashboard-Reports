⚙ Configuration

Enter your Jira connection details. Saved locally in your browser.
⚠️ CORS strategy: The app first tries a direct call to Jira (works for on-prem / Data Center with CORS open). If that fails it falls back to your Cloudflare Worker at {proxyUrl}/jira?jiraBase=…&auth=…&jql=…. The Worker receives jiraBase, auth, jql, fields as plain query params, constructs the Jira search URL server-side, and proxies it — no custom request headers, so no CORS pre-flight fires. Minimal Worker code:

addEventListener('fetch', e => e.respondWith(handle(e.request))) async function handle(req) { if (req.method === 'OPTIONS') return new Response(null, { headers: cors() }) const u = new URL(req.url) if (u.pathname.endsWith('/jira')) { const jiraBase = u.searchParams.get('jiraBase') const auth = u.searchParams.get('auth') const jql = u.searchParams.get('jql') const fields = u.searchParams.get('fields') const maxResults = u.searchParams.get('maxResults') || '100' const startAt = u.searchParams.get('startAt') || '0' if (!jiraBase || !auth) return new Response('Missing jiraBase or auth', { status: 400 }) const target = `${jiraBase}/rest/api/3/search?` + `jql=${encodeURIComponent(jql)}&fields=${fields}` + `&maxResults=${maxResults}&startAt=${startAt}&expand=subtasks` const r = await fetch(target, { headers: { Authorization: 'Basic ' + auth, Accept: 'application/json' } }) return new Response(r.body, { status: r.status, headers: cors(r) }) } return new Response('Not found', { status: 404 }) } function cors(r) { const h = r ? Object.fromEntries(r.headers) : {} h['Access-Control-Allow-Origin'] = '*' h['Access-Control-Allow-Methods'] = 'GET, OPTIONS' h['Access-Control-Allow-Headers'] = 'Accept' return h }
